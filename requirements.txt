# cjet-vision-pipeline 依赖清单
#
# 安装方法（在项目根目录执行）：
#   .conda\python.exe -m pip install -r requirements.txt
#
# 安装后建议再跑一次冒烟测试：
#   .conda\python.exe -c "import supervision, cleanvision, ultralytics; print(supervision.__version__, cleanvision.__version__, ultralytics.__version__)"
#
# 版本说明：下方版本均为本机已验证可运行的组合（2026-11-03 按实际环境核验）。
#   - PyTorch 钉的是 PyPI 默认（CPU）版本；本机实际用的是 CUDA 12.x
#     构建（torch 2.12.1+cu130），如需 CUDA 版见“PyTorch”段注释中的命令。
#     直接 `pip install -r requirements.txt` 会装成 CPU 版并覆盖已有的 CUDA
#     构建——本机有 GPU，请优先走该段给出的 cu130 索引命令。
#   - SAM3 文本提示标注走 ultralytics 的 SAM3SemanticPredictor，仅依赖
#     CLIP（提供 `clip` 包）；权重为本地文件
#     weight/sam3.1_multiplex.pt（不走 PyPI，需自行放置）。不需要
#     transformers —— 那是旧的 facebook/sam3(HF) 方案遗留，现已弃用。
#   - LabelMe 导出由 tools.core 自行实现（supervision 0.29.x 缺失
#     dataset.formats.labelme 模块），无需额外依赖。
# =============================================================================

# ---------- 核心依赖（项目运行时必须） ----------
# 自动标注 / 几何 / 数据集封装都依赖它
supervision==0.29.1
# 模糊目标检测（tools/clean/blurry.py 依赖）
cleanvision==0.3.7

# ---------- 训练 / 推理（推荐） ----------
# YOLOv8 系列训练与推理；同时提供 SAM3 的 SAM3SemanticPredictor
ultralytics==8.4.93
# SAM3 文本提示依赖：ultralytics 的 SAM3SemanticPredictor 需要 CLIP。
# 本机装的是 PyPI 上的 clip（OpenAI 官方包），与 ultralytics 维护的同名仓库
# 分发来源不同；SAM3 只用到 `clip` 这个导入名，两者均可。
clip==1.0

# ---------- PyTorch ----------
# PyPI 默认是 CPU 构建，钉版本以保证可复现：
torch==2.12.1
# torchvision 由 ultralytics / clip 间接依赖，仓库内没有直接 import；
# 保留钉版本是为了锁住与 torch 的 ABI 匹配。
torchvision==0.27.1
# 如需 CUDA 12.x 构建（本机即用此版，推理更快），改用 pytorch 官方源安装：
#   .conda\python.exe -m pip install torch==2.12.1 torchvision==0.27.1 --index-url https://download.pytorch.org/whl/cu130

# ---------- ONNX 推理 ----------
# ONNX 检测器（tools/annotate/backends/onnx.py 通用封装）使用；
# GPU 版优先，无 CUDA 环境时改为 onnxruntime。
onnxruntime-gpu==1.27.0

# ---------- 图像 / 视频处理 ----------
numpy==2.4.4
opencv-python==5.0.0.93
Pillow==12.2.0

# ---------- 工作流 / 工具 ----------
# 进度条（tools/annotate/auto.py 等使用）
tqdm==4.68.4
# 读取 workflow_config.yaml 必需
PyYAML==6.0.3

# ---------- 开发依赖 ----------
# 运行测试（CONTRIBUTING.md 的验证命令依赖它）：pip install -r requirements-dev.txt
# 或直接：pip install pytest
# 见 requirements-dev.txt

# 说明：pycocotools 曾被列为「COCO 标注转换可选依赖」，但仓库内没有任何 import，
# 已从清单移除；如将来实现 COCO 导出再按需加入。
